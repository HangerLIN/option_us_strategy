```python
from prometheus_client import Counter, Histogram, Gauge, generate_latest, CONTENT_TYPE_LATEST
from fastapi import Response

# ============================================================
# 回测指标定义
# ============================================================

# 回测运行统计
BACKTEST_RUNS_TOTAL = Counter(
    "backtest_runs_total",
    "Total number of backtest runs",
    labelnames=("status",)  # success, failed, timeout
)

BACKTEST_DURATION_SECONDS = Histogram(
    "backtest_duration_seconds",
    "Backtest execution duration",
    labelnames=("strategy",),
    buckets=(10, 30, 60, 120, 300, 600, 1800, 3600)
)

BACKTEST_ACTIVE_RUNS = Gauge(
    "backtest_active_runs",
    "Number of currently running backtests"
)

# API请求统计
BACKTEST_API_REQUESTS_TOTAL = Counter(
    "backtest_api_requests_total",
    "Total API requests",
    labelnames=("endpoint", "status_code")
)

BACKTEST_API_LATENCY_SECONDS = Histogram(
    "backtest_api_latency_seconds",
    "API request latency",
    labelnames=("endpoint",),
    buckets=(0.01, 0.05, 0.1, 0.5, 1.0, 2.0, 5.0)
)

# 数据质量指标
BACKTEST_DATA_COVERAGE_PCT = Gauge(
    "backtest_data_coverage_pct",
    "Data coverage percentage for backtest period",
    labelnames=("run_id",)
)

BACKTEST_TRADE_COUNT = Gauge(
    "backtest_trade_count",
    "Number of trades in backtest",
    labelnames=("run_id", "symbol")
)

# 性能指标
BACKTEST_PNL_USD = Gauge(
    "backtest_pnl_usd",
    "Backtest P&L in USD",
    labelnames=("run_id", "metric_type")  # realized, unrealized, total
)

BACKTEST_WIN_RATE_PCT = Gauge(
    "backtest_win_rate_pct",
    "Win rate percentage",
    labelnames=("run_id",)
)

BACKTEST_SHARPE_RATIO = Gauge(
    "backtest_sharpe_ratio",
    "Sharpe ratio",
    labelnames=("run_id",)
)

# ============================================================
# Prometheus端点
# ============================================================

@app.get(
    "/metrics",
    summary="Prometheus指标",
    description="暴露Prometheus格式的监控指标",
    include_in_schema=False  # 不在OpenAPI文档中显示
)
async def prometheus_metrics():
    """返回Prometheus格式的指标数据"""
    return Response(
        content=generate_latest(),
        media_type=CONTENT_TYPE_LATEST
    )

# ============================================================
# 中间件：记录API请求指标
# ============================================================

import time
from fastapi import Request

@app.middleware("http")
async def metrics_middleware(request: Request, call_next):
    """记录每个请求的指标"""
    start_time = time.time()
    
    response = await call_next(request)
    
    duration = time.time() - start_time
    endpoint = request.url.path
    status_code = str(response.status_code)
    
    BACKTEST_API_REQUESTS_TOTAL.labels(
        endpoint=endpoint,
        status_code=status_code
    ).inc()
    
    BACKTEST_API_LATENCY_SECONDS.labels(
        endpoint=endpoint
    ).observe(duration)
    
    return response
```

#### 1.2 在回测执行时记录指标

修改回测运行逻辑（假设在某个回测执行函数中）：

```python
import time

def execute_backtest(run_id: int, config: dict):
    """执行回测并记录指标"""
    BACKTEST_ACTIVE_RUNS.inc()
    start_time = time.time()
    status = "unknown"
    
    try:
        # 执行回测逻辑
        result = run_backtest(...)
        
        # 记录成功
        status = "success"
        BACKTEST_RUNS_TOTAL.labels(status="success").inc()
        
        # 记录指标
        if result:
            BACKTEST_PNL_USD.labels(
                run_id=str(run_id),
                metric_type="total"
            ).set(result.total_pnl)
            
            BACKTEST_WIN_RATE_PCT.labels(
                run_id=str(run_id)
            ).set(result.win_rate * 100)
            
            BACKTEST_TRADE_COUNT.labels(
                run_id=str(run_id),
                symbol=config.get("symbol", "UNKNOWN")
            ).set(result.trade_count)
        
    except TimeoutError:
        status = "timeout"
        BACKTEST_RUNS_TOTAL.labels(status="timeout").inc()
        raise
        
    except Exception as e:
        status = "failed"
        BACKTEST_RUNS_TOTAL.labels(status="failed").inc()
        raise
        
    finally:
        duration = time.time() - start_time
        BACKTEST_DURATION_SECONDS.labels(
            strategy=config.get("strategy", "unknown")
        ).observe(duration)
        
        BACKTEST_ACTIVE_RUNS.dec()
        
        LOGGER.info(
            "Backtest completed",
            run_id=run_id,
            status=status,
            duration=duration
        )
```

创建 `infra/prometheus/rules/backtest.yml`：

```yaml
groups:
  - name: backtest_alerts
    interval: 60s
    rules:
      # 回测失败率过高
      - alert: HighBacktestFailureRate
        expr: |
          (
            rate(backtest_runs_total{status="failed"}[1h])
            /
            rate(backtest_runs_total[1h])
          ) > 0.2
        for: 10m
        labels:
          severity: medium
          service: backtest_api
          category: quality
        annotations:
          summary: "回测失败率过高"
          description: "过去1小时回测失败率: {{ $value | humanizePercentage }}"
          runbook: "1) 检查数据完整性 2) 查看回测日志 3) 检查数据库连接"

      # 回测耗时过长
      - alert: BacktestExecutionSlow
        expr: |
          histogram_quantile(0.95,
            rate(backtest_duration_seconds_bucket[30m])
          ) > 1800
        for: 30m
        labels:
          severity: low
          service: backtest_api
          category: performance
        annotations:
          summary: "回测执行耗时过长"
          description: "P95回测耗时: {{ $value | humanize }}s (>30分钟)"

      # 回测API响应慢

        expr: |
          histogram_quantile(0.95,
            rate(backtest_api_latency_seconds_bucket[10m])
          ) > 5
        for: 10m
        labels:
          severity: medium
          service: backtest_api
          category: performance
        annotations:
          summary: "回测API响应缓慢"
          description: "P95延迟: {{ $value | humanize }}s"

      # 回测API错误率高
      - alert: HighBacktestAPIErrorRate
        expr: |
          (
            sum(rate(backtest_api_requests_total{status_code=~"5.."}[5m]))
            /
            sum(rate(backtest_api_requests_total[5m]))
          ) > 0.05
        for: 5m
        labels:
          severity: high
          service: backtest_api
          category: reliability
        annotations:
          summary: "回测API错误率高"
          description: "错误率: {{ $value | humanizePercentage }}"

      # 没有回测运行（工作时间）
      - alert: NoBacktestsRunning

          (
            rate(backtest_runs_total[2h]) == 0
```

## 📊 简化版实现（最小可行方案）

如果时间紧张，可以先实现最基本的监控：

### 最小实施清单

2. ✅ 回测运行计数 (成功/失败)
3. ✅ 回测耗时统计

**可以延后**:
- 详细的回测质量指标
- 复杂的告警规则
- 专属Grafana看板

### 最小代码修改

只需在 `apps/backtest/api.py` 顶部添加：

```python
from prometheus_client import Counter, Histogram, generate_latest, CONTENT_TYPE_LATEST
from fastapi import Response
import time

# 基本指标
backtest_requests = Counter('backtest_api_requests_total', 'Total requests', ['endpoint'])
backtest_latency = Histogram('backtest_api_latency_seconds', 'Request latency', ['endpoint'])

@app.middleware("http")
async def track_metrics(request: Request, call_next):
    start = time.time()
    response = await call_next(request)
    backtest_latency.labels(endpoint=request.url.path).observe(time.time() - start)
    backtest_requests.labels(endpoint=request.url.path).inc()
    return response

@app.get("/metrics", include_in_schema=False)
async def metrics():
    return Response(content=generate_latest(), media_type=CONTENT_TYPE_LATEST)
```

然后更新Prometheus配置添加target即可。

## 💡 最佳实践

1. **回测指标保留策略**
   - 基础指标（计数、延迟）: 30天
   - 详细指标（per-run metrics）: 7天
   - 使用标签而不是创建太多指标

2. **避免高基数标签**
   - ❌ 不要用 run_id 作为标签（会爆炸）
   - ✅ 用 status、strategy 等低基数标签
   - 详细数据查数据库，监控只看聚合趋势

3. **回测与实盘的区分**
   - 所有回测指标加 `backtest_` 前缀
   - 实盘指标保持现有命名
   - 看板上清晰区分回测和实盘数据
